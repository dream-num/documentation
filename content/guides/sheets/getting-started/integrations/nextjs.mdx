---
title: Integrate in Next.js
---

Before you begin, read [installation and basic usage](/guides/sheets/getting-started/installation). The example mounts Univer Sheets in a container with an explicit height, using the Next.js App Router.

Load the editor from a Client Component with `ssr: false`[^ssr]. A `use client` directive alone does not disable prerendering. See [Next.js: skipping SSR](https://nextjs.org/docs/app/guides/lazy-loading#skipping-ssr) for details.

### Integration steps

1. Initialize Univer inside `useEffect`.
2. Dispose of Univer in the cleanup function returned by `useEffect`.

### Preset configuration [#preset-configuration]

`UniverSheetsCorePreset`

| <span data-config-options="">Option</span> | Type | Default | What it does |
| --- | --- | --- | --- |
| `container` | `string \| HTMLElement` | `'app'` | The element into which Univer is mounted, or the DOM ID of that element. |

### Example

Each mount uses its own child container. Cleanup defers disposal to a microtask, by which point React has finished its own cleanup, so nested React roots are never unmounted synchronously during a React commit.

```tsx title="univer-editor.tsx"
'use client'

import { UniverSheetsCorePreset } from '@univerjs/preset-sheets-core'
import UniverPresetSheetsCoreEnUS from '@univerjs/preset-sheets-core/locales/en-US'
import { createUniver, LocaleType, mergeLocales } from '@univerjs/presets'
import { useEffect, useRef } from 'react'

import '@univerjs/preset-sheets-core/lib/index.css'

export default function UniverEditor() {
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const host = containerRef.current
    if (!host) return
    const container = document.createElement('div')
    container.style.height = '100%'
    host.append(container)

    const { univer, univerAPI } = createUniver({
      locale: LocaleType.EN_US,
      locales: {
        [LocaleType.EN_US]: mergeLocales(
          UniverPresetSheetsCoreEnUS,
        ),
      },
      presets: [
        UniverSheetsCorePreset({
          container,
        }),
      ],
    })

    univerAPI.createWorkbook({})

    return () => {
      queueMicrotask(() => {
        univer.dispose()
        container.remove()
      })
    }
  }, [])

  return (
    <div ref={containerRef} style={{ height: 600 }} />
  )
}
```

```tsx title="page.tsx"
'use client'

import dynamic from 'next/dynamic'

const UniverEditor = dynamic(() => import('./univer-editor'), { ssr: false })

export default function Page() {
  return <UniverEditor />
}
```

[^ssr]: Server-side rendering (SSR) runs rendering code before the browser loads the page. Browser-only APIs such as `window` and the DOM are unavailable there; a Client Component may still be prerendered unless SSR is disabled for that editor component.
