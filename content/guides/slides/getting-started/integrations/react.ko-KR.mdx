title: React에 통합하기

먼저 [설치 및 기본 사용](/guides/slides/getting-started/installation)를 읽으십시오. React 부품의 Univer 슬라이드를 장착하고 컨테이너에 명시된 높이를 설정합니다.

### 통합 단계

1. `useEffect`에서 Univer를 초기화합니다.
2. `useEffect`의 정리 함수에서 인스턴스를 해제합니다.

### 플러그인 설정 [#plugin-configuration]

`UniverUIPlugin`

| <span data-config-options="">옵션</span> | 유형 | 기본값 | 동작 |
| --- | --- | --- | --- |
| `container` | `string \| HTMLElement` |  | 편집기를 마운트할 DOM 요소 또는 요소 ID입니다. |

`UniverLicensePlugin`

| <span data-config-options="">옵션</span> | 유형 | 기본값 | 동작 |
| --- | --- | --- | --- |
| `license` | `string` |  | 클라이언트 라이선스 파일의 내용입니다. |

### 예제 코드

각 마운트는 자체 아이 컨테이너를받습니다. 처리는 React가 주변의 정리를 완료 한 후 microtask에서 실행되므로 React 루트가 React 커밋 중에 비동기적으로 배치되지 않습니다.

```tsx
import { useEffect, useRef } from 'react'
import { UniverLicensePlugin } from '@univerjs-pro/license'
import { UniverSlidesPlugin } from '@univerjs-pro/slides'
import { UniverSlidesUIPlugin } from '@univerjs-pro/slides-ui'
import SlidesUIkoKR from '@univerjs-pro/slides-ui/locale/ko-KR'
import SlideskoKR from '@univerjs-pro/slides/locale/ko-KR'
import { LocaleType, mergeLocales, Univer, UniverInstanceType } from '@univerjs/core'
import DesignkoKR from '@univerjs/design/locale/ko-KR'
import { UniverDocsPlugin } from '@univerjs/docs'
import { UniverDocsUIPlugin } from '@univerjs/docs-ui'
import DocsUIkoKR from '@univerjs/docs-ui/locale/ko-KR'
import { UniverDrawingPlugin } from '@univerjs/drawing'
import { UniverRenderEnginePlugin } from '@univerjs/engine-render'
import { UniverUIPlugin } from '@univerjs/ui'
import UIkoKR from '@univerjs/ui/locale/ko-KR'

import '@univerjs/design/lib/index.css'
import '@univerjs/ui/lib/index.css'
import '@univerjs/docs-ui/lib/index.css'
import '@univerjs-pro/slides-ui/lib/index.css'

export function UniverEditor() {
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const host = containerRef.current
    if (!host) return
    const container = document.createElement('div')
    container.style.height = '100%'
    host.append(container)

    const univer = new Univer({
      locale: LocaleType.KO_KR,
      locales: {
        [LocaleType.KO_KR]: mergeLocales(
          DesignkoKR,
          UIkoKR,
          DocsUIkoKR,
          SlideskoKR,
          SlidesUIkoKR,
        ),
      },
    })

    univer.registerPlugin(UniverLicensePlugin, {
      license: 'your license.txt',
    })
    univer.registerPlugin(UniverRenderEnginePlugin)
    univer.registerPlugin(UniverUIPlugin, {
      container,
    })

    univer.registerPlugin(UniverDocsPlugin)
    univer.registerPlugin(UniverDocsUIPlugin)
    univer.registerPlugin(UniverDrawingPlugin)


    univer.registerPlugin(UniverSlidesPlugin)
    univer.registerPlugin(UniverSlidesUIPlugin)

    univer.createUnit(UniverInstanceType.UNIVER_SLIDE, {})

    return () => {
      queueMicrotask(() => {
        univer.dispose()
        container.remove()
      })
    }
  }, [])

  return <div ref={containerRef} style={{ height: 600 }} />
}
```
