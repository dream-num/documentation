---
title: Intégrer dans Astro
---

Consultez d’abord [l’installation et les premiers pas](/guides/slides/getting-started/installation). Cet exemple utilise Astro pour monter Univer Slides dans un conteneur avec une hauteur explicite.

Utilisez un `<script>` normal pour initialiser directement l’éditeur dans le navigateur. Initialisez au premier chargement et sur `astro:page-load`, puis libérez l’instance sur `astro:before-swap` avec ClientRouter. N’importez pas l’éditeur dans le frontmatter Astro. Consultez les [événements de navigation Astro](https://docs.astro.build/en/reference/modules/astro-transitions/).

### Étapes d’intégration

1. Initialisez Univer dans `mount()` / `astro:page-load`.
2. Libérez Univer dans `astro:before-swap`.

### Exemple de code

```astro title="src/pages/index.astro"
<div id="univer-slides" style="height: 600px"></div>

<script>
  import { UniverLicensePlugin } from '@univerjs-pro/license'
  import { UniverSlidesPlugin } from '@univerjs-pro/slides'
  import { UniverSlidesUIPlugin } from '@univerjs-pro/slides-ui'
  import SlidesUIfrFR from '@univerjs-pro/slides-ui/locale/fr-FR'
  import SlidesfrFR from '@univerjs-pro/slides/locale/fr-FR'
  import { LocaleType, mergeLocales, Univer, UniverInstanceType } from '@univerjs/core'
  import DesignfrFR from '@univerjs/design/locale/fr-FR'
  import { UniverDocsPlugin } from '@univerjs/docs'
  import { UniverDocsUIPlugin } from '@univerjs/docs-ui'
  import DocsUIfrFR from '@univerjs/docs-ui/locale/fr-FR'
  import { UniverDrawingPlugin } from '@univerjs/drawing'
  import { UniverRenderEnginePlugin } from '@univerjs/engine-render'
  import { UniverUIPlugin } from '@univerjs/ui'
  import UIfrFR from '@univerjs/ui/locale/fr-FR'

  import '@univerjs/design/lib/index.css'
  import '@univerjs/ui/lib/index.css'
  import '@univerjs/docs-ui/lib/index.css'
  import '@univerjs-pro/slides-ui/lib/index.css'

  let univerInstance: Univer | undefined

  function mount() {
    const container = document.getElementById('univer-slides')
    if (!container || univerInstance) return

    const univer = new Univer({
      locale: LocaleType.FR_FR,
      locales: {
        [LocaleType.FR_FR]: mergeLocales(
          DesignfrFR,
          UIfrFR,
          DocsUIfrFR,
          SlidesfrFR,
          SlidesUIfrFR,
        ),
      },
    })

    univer.registerPlugin(UniverLicensePlugin, {
      license: 'your license.txt',
    })
    univer.registerPlugin(UniverRenderEnginePlugin)
    univer.registerPlugin(UniverUIPlugin, {
      container,
    })

    univer.registerPlugin(UniverDocsPlugin)
    univer.registerPlugin(UniverDocsUIPlugin)
    univer.registerPlugin(UniverDrawingPlugin)


    univer.registerPlugin(UniverSlidesPlugin)
    univer.registerPlugin(UniverSlidesUIPlugin)

    univer.createUnit(UniverInstanceType.UNIVER_SLIDE, {})
    univerInstance = univer
  }

  function dispose() {
    univerInstance?.dispose()
    univerInstance = undefined
  }

  mount()
  document.addEventListener('astro:page-load', mount)
  document.addEventListener('astro:before-swap', dispose)
</script>
```
