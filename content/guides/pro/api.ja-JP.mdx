## ドキュメントを作成

<APITable
  request={{
    method: 'POST',
    url: '/universer-api/snapshot/{type}/unit/-/create',
    headers: <code>content-type: application/json</code>,
    parametersType: 'Body',
    parameters: [{
      name: 'type',
      type: 'enum(int)',
      required: true,
      description: '1（docs）、2（sheets）のパスとリクエストボディのパラメータ',
    }, {
      name: 'name',
      type: 'string',
      required: true,
      description: 'ドキュメント名',
    }, {
      name: 'creator',
      type: 'string',
      required: true,
      description: '作成者 ID',
    }],
    example: `curl http://localhost:8000/universer-api/snapshot/{type}/unit/-/create \\
    -X POST \\
    -H 'Content-Type: application/json' \\
    --data-raw '{"type":2,"name":"New Sheet By Univer","creator":"userID"}'`,
  }}
  response={{
    type: 'application/json',
    parameters: [{
      name: 'error',
      type: 'object',
      properties: [{
        name: 'code',
        type: 'enum(int)',
        description: '1（成功）',
      }, {
        name: 'message',
        type: 'string',
        description: '失敗情報',
      }],
    }, {
      name: 'unitID',
      type: 'string',
      description: 'ドキュメント ID',
    }],
    example: JSON.stringify({
      error: {
        code: 1,
        message: 'success',
      },
      unitID: 'ETVf-B4lQqOSE_p09mcp9Q',
    }, null, 2),
  }}
/>

## ドキュメント一覧を取得

<APITable
  request={{
    method: 'GET',
    url: '/universer-api/snapshot/{type}/units',
    parametersType: 'Query',
    parameters: [{
      name: 'type',
      type: 'enum(int)',
      required: true,
      description: '1（doc）、2（sheet）のパスとリクエストボディのパラメータ',
    }, {
      name: 'nextCursor',
      type: 'string',
      required: false,
      description: 'ページネーションの次ページカーソル',
    }],
    example: `curl -X GET 'http://localhost:8000/universer-api/snapshot/1/units?nextCursor=100'`,
  }}
  response={{
    type: 'application/json',
    parameters: [{
      name: 'error',
      type: 'object',
      properties: [{
        name: 'code',
        type: 'enum(int)',
        description: '1（成功）',
      }, {
        name: 'message',
        type: 'string',
        description: '失敗情報',
      }],
    }, {
      name: 'units',
      type: 'array[object]',
      description: 'ドキュメント一覧',
      properties: [{
        name: 'unitID',
        type: 'string',
        description: 'ドキュメント ID',
      }, {
        name: 'name',
        type: 'string',
        description: 'ドキュメント名',
      }, {
        name: 'type',
        type: 'enum(int)',
        description: '1（doc）、2（sheet）',
      }],
    }, {
      name: 'nextCursor',
      type: 'string',
      description: 'ページネーションの次ページカーソル。空の場合は最終ページ',
    }],
    example: JSON.stringify({
      error: {
        code: 1,
        message: '',
      },
      units: [{
        unitID: '1',
        name: 'a',
        type: 1,
      }],
      nextCursor: '200',
    }, null, 2),
  }}
/>

## ファイルをアップロード

<APITable
  request={{
    method: 'POST',
    url: '/universer-api/stream/file/upload',
    headers: <code>content-type: multipart/form-data</code>,
    parametersType: 'Body',
    parameters: [{
      name: 'size',
      type: 'int',
      required: true,
      description: 'ファイルサイズ（バイト）。クエリパラメータ。実際のサイズと一致しない場合はエラーになります。',
    }, {
      name: 'file',
      type: 'Form.file',
      required: true,
      description: 'HTML フォームのファイルアップロード',
    }],
    example: `curl 'http://localhost:8000/universer-api/stream/file/upload?size=125466' \\
    --header 'cookie: _univer=XXXXXX' \\
    --form 'file=@"demo.xlsx"'`,
  }}
  response={{
    type: 'application/json',
    parameters: [{
      name: 'FileId',
      type: 'string',
      description: 'ファイル ID。インポート/エクスポートなど後続のリクエストで使用',
    }],
    example: JSON.stringify({
      FileId: 'xxxx',
    }, null, 2),
  }}
/>

    method: 'POST',

    parametersType: 'Body',
    parameters: [{
      name: 'type',
      type: 'enum(int)',
      required: true,
      description: '1（doc）、2（sheet）',
    }, {
      name: 'outputType',
      type: 'enum(int)',
      required: true,
      description: '1（unit）、2（json）',

      name: 'fileID',
      type: 'string',

    }, {
      name: 'minSheetRowCount',
      type: 'int',
      required: true,
      description: '最小行数。インポート時にシートの行数を確認します。',
    }, {
      name: 'minSheetColumnCount',
      type: 'int',
      required: true,
      description: '最小列数。インポート時にシートの列数を確認します。',
    }],
    example: `curl -X POST 'http://localhost:8000/universer-api/exchange/2/import' \\
    -H 'Content-Type: application/json' \\
    --data-raw '{"fileID":"123","outputType":1,"minSheetRowCount":1000,"minSheetColumnCount":20}'`,
  }}
  response={{
    type: 'application/json',
    parameters: [{
      name: 'error',
      type: 'object',

        name: 'code',
        type: 'enum(int)',
        description: '1（成功）',
      }, {
        name: 'message',
        type: 'string',
        description: '失敗情報',
      }],
    }, {
      name: 'taskID',
      type: 'string',
      description: '変換タスク ID。インポート/エクスポートは非同期のため、この ID を使って結果をポーリングします。',
    }],

      error: {
        code: 1,
        message: '',

      taskID: '456',

  }}

## 変換結果を取得

<APITable
  request={{
    method: 'GET',
    url: '/universer-api/exchange/task/{taskID}',
    parametersType: 'Path',
    parameters: [{
      name: 'taskID',
      type: 'string',
      required: true,
      description: '変換タスク ID',
    }],
    example: `curl -X GET 'http://localhost:8000/universer-api/exchange/task/123'`,
  }}
  response={{
    type: 'application/json',
    parameters: [{
      name: 'error',
      type: 'object',
      properties: [{
        name: 'code',
        type: 'enum(int)',
        description: '1（成功）',
      }, {
        name: 'message',
        type: 'string',
        description: '失敗情報',
      }],
    }, {
      name: 'status',
      type: 'enum(string)',
      description: '"pending", "done", "failed"',
    }, {
      name: 'export',
      type: 'object',
      properties: [{
        name: 'fileID',
        type: 'string',
        description: 'エクスポート変換結果のファイル ID。ファイル内容取得の後続リクエストで使用',
      }],
    }, {
      name: 'import',
      type: 'object',
      properties: [{
        name: 'unitID',
        type: 'string',
        description: 'インポート後の unit ID。オンラインスプレッドシートを開くために使用',
      }, {
        name: 'jsonID',
        type: 'string',
        description: 'JSON 形式のインポート変換結果のファイル ID。ファイル内容取得の後続リクエストで使用',
      }],
    }],
    example: JSON.stringify({
      error: {
        code: 1,
        message: '',
      },
      status: 'done',
      export: {
        fileID: '456',
      },
      import: {
        unitID: '789',
        jsonID: '012',
      },
    }, null, 2),
  }}
