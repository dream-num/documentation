## 建立文件

<APITable
  request={{
    method: 'POST',
    url: '/universer-api/snapshot/{type}/unit/-/create',
    headers: <code>content-type: application/json</code>,
    parametersType: 'Body',
    parameters: [{
      name: 'type',
      type: 'enum(int)',
      required: true,
      description: '1（docs），2（sheets）路徑和請求體引數',
    }, {
      name: 'name',
      type: 'string',
      required: true,
      description: '文件名稱',
    }, {
      name: 'creator',
      type: 'string',
      required: true,
      description: '建立者 id',
    }],
    example: `curl http://localhost:8000/universer-api/snapshot/{type}/unit/-/create \\
    -X POST \\
    -H 'Content-Type: application/json' \\
    --data-raw '{"type":2,"name":"New Sheet By Univer","creator":"userID"}'`,
  }}
  response={{
    type: 'application/json',
    parameters: [{
      name: 'error',
      type: 'object',
      properties: [{
        name: 'code',
        type: 'enum(int)',
        description: '1（成功）',
      }, {
        name: 'message',
        type: 'string',
        description: '失敗資訊',
      }],
    }, {
      name: 'unitID',
      type: 'string',
      description: '文件 id',
    }],
    example: JSON.stringify({
      error: {
        code: 1,
        message: 'success',
      },
      unitID: 'ETVf-B4lQqOSE_p09mcp9Q',
    }, null, 2),
  }}
/>

## 獲取文件列表

<APITable
  request={{
    method: 'GET',
    url: '/universer-api/snapshot/{type}/units',
    parametersType: 'Query',
    parameters: [{
      name: 'type',
      type: 'enum(int)',
      required: true,
      description: '1（doc），2（sheet）路徑和請求體引數',
    }, {
      name: 'nextCursor',
      type: 'string',
      required: false,
      description: '分頁的下一頁遊標',
    }],
    example: `curl -X GET 'http://localhost:8000/universer-api/snapshot/1/units?nextCursor=100'`,
  }}
  response={{
    type: 'application/json',
    parameters: [{
      name: 'error',
      type: 'object',
      properties: [{
        name: 'code',
        type: 'enum(int)',
        description: '1（成功）',
      }, {
        name: 'message',
        type: 'string',
        description: '失敗資訊',
      }],
    }, {
      name: 'units',
      type: 'array[object]',
      description: '文件列表',
      properties: [{
        name: 'unitID',
        type: 'string',
        description: '文件 id',
      }, {
        name: 'name',
        type: 'string',
        description: '文件名稱',
      }, {
        name: 'type',
        type: 'enum(int)',
        description: '1（doc），2（sheet）',
      }],
    }, {
      name: 'nextCursor',
      type: 'string',
      description: '分頁的下一頁遊標，當為空時表示為最後一頁',
    }],
    example: JSON.stringify({
      error: {
        code: 1,
        message: '',
      },
      units: [{
        unitID: '1',
        name: 'a',
        type: 1,
      }],
      nextCursor: '200',
    }, null, 2),
  }}
/>

## 上傳檔案

<APITable
  request={{
    method: 'POST',
    url: '/universer-api/stream/file/upload',
    headers: <code>content-type: multipart/form-data</code>,
    parametersType: 'Body',
    parameters: [{
      name: 'size',
      type: 'int',
      required: true,
      description: '檔案大小（byte），query 引數；引數必須等於真實檔案大小，如果不等於會造成返回 error',
    }, {
      name: 'file',
      type: 'Form.file',
      required: true,
      description: 'HTML form 上傳檔案',
    }],
    example: `curl 'http://localhost:8000/universer-api/stream/file/upload?size=125466' \\
    --header 'cookie: _univer=XXXXXX' \\
    --form 'file=@"demo.xlsx"'`,
  }}
  response={{
    type: 'application/json',
    parameters: [{
      name: 'FileId',
      type: 'string',
      description: '檔案 id，用於後面請求比如匯入和匯出',
    }],
    example: JSON.stringify({
      FileId: 'xxxx',
    }, null, 2),
  }}
/>

## 匯入

<APITable
  request={{
    method: 'POST',
    url: '/universer-api/exchange/{type}/import',
    headers: <code>content-type: application/json</code>,
    parametersType: 'Body',
    parameters: [{
      name: 'type',
      type: 'enum(int)',
      required: true,
      description: '1（doc），2（sheet）',
    }, {
      name: 'outputType',
      type: 'enum(int)',
      required: true,
      description: '1（unit），2（json）',
    }, {
      name: 'fileID',
      type: 'string',
      required: true,
      description: '上傳的檔案 id',
    }, {
      name: 'minSheetRowCount',
      type: 'int',
      required: true,
      description: '最小行數，匯入時會檢查表格行數是否滿足要求',
    }, {
      name: 'minSheetColumnCount',
      type: 'int',
      required: true,
      description: '最小列數，匯入時會檢查表格列數是否滿足要求',
    }],
    example: `curl -X POST 'http://localhost:8000/universer-api/exchange/2/import' \\
    -H 'Content-Type: application/json' \\
    --data-raw '{"fileID":"123","outputType":1,"minSheetRowCount":1000,"minSheetColumnCount":20}'`,
  }}
  response={{
    type: 'application/json',
    parameters: [{
      name: 'error',
      type: 'object',
      properties: [{
        name: 'code',
        type: 'enum(int)',
        description: '1（成功）',
      }, {
        name: 'message',
        type: 'string',
        description: '失敗資訊',
      }],
    }, {
      name: 'taskID',
      type: 'string',
      description: '轉換任務 id，匯入匯出是非同步的，使用該 id 輪詢介面獲取轉換結果',
    }],
    example: JSON.stringify({
      error: {
        code: 1,
        message: '',
      },
      taskID: '456',
    }, null, 2),
  }}
/>

## 獲取轉換結果

<APITable
  request={{
    method: 'GET',
    url: '/universer-api/exchange/task/{taskID}',
    parametersType: 'Path',
    parameters: [{
      name: 'taskID',
      type: 'string',
      required: true,
      description: '轉換任務 id',
    }],
    example: `curl -X GET 'http://localhost:8000/universer-api/exchange/task/123'`,
  }}
  response={{
    type: 'application/json',
    parameters: [{
      name: 'error',
      type: 'object',
      properties: [{
        name: 'code',
        type: 'enum(int)',
        description: '1（成功）',
      }, {
        name: 'message',
        type: 'string',
        description: '失敗資訊',
      }],
    }, {
      name: 'status',
      type: 'enum(string)',
      description: '"pending", "done", "failed"',
    }, {
      name: 'export',
      type: 'object',
      properties: [{
        name: 'fileID',
        type: 'string',
        description: '匯出轉換結果的檔案 id，用於後面介面獲取檔案內容',
      }],
    }, {
      name: 'import',
      type: 'object',
      properties: [{
        name: 'unitID',
        type: 'string',
        description: '匯入轉換後的 unitID，用於開啟線上表格',
      }, {
        name: 'jsonID',
        type: 'string',
        description: '匯入轉換為 JSON 結果的檔案 id，用於後面介面獲取檔案內容',
      }],
    }],
    example: JSON.stringify({
      error: {
        code: 1,
        message: '',
      },
      status: 'done',
      export: {
        fileID: '456',
      },
      import: {
        unitID: '789',
        jsonID: '012',
      },
    }, null, 2),
  }}
/>
