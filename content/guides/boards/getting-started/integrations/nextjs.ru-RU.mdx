---
title: Интеграция с Next.js
---

Сначала прочитайте [Установка и базовое использование](/guides/boards/getting-started/installation). В этом примере монтируются платы Univer из компонента клиента маршрутизатора приложения Next.js в контейнер с явной высотой.

Загрузить редактор из Компонента клиента с помощью `ssr: false`[^ssr]. Добавление `use client` не отключает саму рендеринг. См. [Пропуск SSR в Next.js](https://nextjs.org/docs/app/guides/lazy-loading#skipping-ssr).

Заменить `YOUR_CLIENT_LICENSE` на содержимое лицензии клиента, а не на путь файла. См. [Конфигурация лицензии](/server/license).

### Шаги интеграции

1. Инициализируйте Univer в `useEffect`.
2. Освободите Univer в `useEffect cleanup`.

### Настройка плагинов [#plugin-configuration]

`UniverUIPlugin`

| <span data-config-options="">Параметр</span> | Тип | По умолчанию | Действие |
| --- | --- | --- | --- |
| `container` | `string \| HTMLElement` |  | DOM-элемент для размещения редактора или его идентификатор. |

`UniverLicensePlugin`

| <span data-config-options="">Параметр</span> | Тип | По умолчанию | Действие |
| --- | --- | --- | --- |
| `license` | `string` |  | Содержимое файла клиентской лицензии. |

### Пример кода

При каждом монтировании создаётся отдельный дочерний контейнер. Экземпляр освобождается в микрозадаче после завершения очистки внешнего компонента React. Это предотвращает синхронное размонтирование вложенных корней React во время коммита.

```tsx title="univer-editor.tsx"
'use client'

import { useEffect, useRef } from 'react'
import { UniverBoardsPlugin } from '@univerjs-pro/boards'
import { UniverBoardsUIPlugin } from '@univerjs-pro/boards-ui'
import BoardsUIruRU from '@univerjs-pro/boards-ui/locale/ru-RU'
import { UniverInkUIPlugin } from '@univerjs-pro/ink-ui'
import InkUIruRU from '@univerjs-pro/ink-ui/locale/ru-RU'
import { UniverLicensePlugin } from '@univerjs-pro/license'
import { LocaleType, mergeLocales, Univer, UniverInstanceType } from '@univerjs/core'
import DesignruRU from '@univerjs/design/locale/ru-RU'
import { UniverDocsPlugin } from '@univerjs/docs'
import { UniverDocsUIPlugin } from '@univerjs/docs-ui'
import DocsUIruRU from '@univerjs/docs-ui/locale/ru-RU'
import { UniverDrawingPlugin } from '@univerjs/drawing'
import { UniverDrawingUIPlugin } from '@univerjs/drawing-ui'
import DrawingUIruRU from '@univerjs/drawing-ui/locale/ru-RU'
import { UniverRenderEnginePlugin } from '@univerjs/engine-render'
import { UniverUIPlugin } from '@univerjs/ui'
import UIruRU from '@univerjs/ui/locale/ru-RU'

import '@univerjs/design/lib/index.css'
import '@univerjs/ui/lib/index.css'
import '@univerjs/docs-ui/lib/index.css'
import '@univerjs/drawing-ui/lib/index.css'
import '@univerjs-pro/ink-ui/lib/index.css'
import '@univerjs-pro/boards-ui/lib/index.css'

export default function UniverEditor() {
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const host = containerRef.current
    if (!host) return
    const container = document.createElement('div')
    container.style.height = '100%'
    host.append(container)

    const univer = new Univer({
      locale: LocaleType.RU_RU,
      locales: {
        [LocaleType.RU_RU]: mergeLocales(
          DesignruRU,
          UIruRU,
          DocsUIruRU,
          DrawingUIruRU,
          InkUIruRU,
          BoardsUIruRU,
        ),
      },
    })

    univer.registerPlugin(UniverLicensePlugin, {
      license: 'YOUR_CLIENT_LICENSE',
    })
    univer.registerPlugin(UniverRenderEnginePlugin)
    univer.registerPlugin(UniverUIPlugin, { container })
    univer.registerPlugin(UniverDocsPlugin)
    univer.registerPlugin(UniverDocsUIPlugin)
    univer.registerPlugin(UniverDrawingPlugin)
    univer.registerPlugin(UniverDrawingUIPlugin)
    univer.registerPlugin(UniverInkUIPlugin)
    univer.registerPlugin(UniverBoardsPlugin)
    univer.registerPlugin(UniverBoardsUIPlugin)

    univer.createUnit(UniverInstanceType.UNIVER_BOARD, {})

    return () => {
      queueMicrotask(() => {
        univer.dispose()
        container.remove()
      })
    }
  }, [])

  return <div ref={containerRef} style={{ height: 600 }} />
}
```

```tsx title="page.tsx"
'use client'

import dynamic from 'next/dynamic'

const UniverEditor = dynamic(() => import('./univer-editor'), { ssr: false })

export default function Page() {
  return <UniverEditor />
}
```

[^ssr]: Серверный рендеринг (SSR) выполняет код отрисовки до загрузки страницы браузером, без `window` и DOM. Client Component также может предварительно отрисовываться, поэтому SSR редактора отключается отдельно.
