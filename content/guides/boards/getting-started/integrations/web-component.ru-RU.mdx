---
title: Интеграция в Web Component
---

Сначала прочитайте [установку и базовую настройку](/guides/boards/getting-started/installation). В этом примере Web Component монтирует Univer Boards в контейнер с явно заданной высотой.

### Шаги интеграции

1. Инициализируйте Univer в `connectedCallback`.
2. Освободите Univer в `disconnectedCallback`.

### Пример кода

```typescript
import { UniverBoardsPlugin } from '@univerjs-pro/boards'
import { UniverBoardsUIPlugin } from '@univerjs-pro/boards-ui'
import BoardsUIruRU from '@univerjs-pro/boards-ui/locale/ru-RU'
import { UniverInkUIPlugin } from '@univerjs-pro/ink-ui'
import InkUIruRU from '@univerjs-pro/ink-ui/locale/ru-RU'
import { UniverLicensePlugin } from '@univerjs-pro/license'
import { LocaleType, mergeLocales, Univer, UniverInstanceType } from '@univerjs/core'
import DesignruRU from '@univerjs/design/locale/ru-RU'
import { UniverDocsPlugin } from '@univerjs/docs'
import { UniverDocsUIPlugin } from '@univerjs/docs-ui'
import DocsUIruRU from '@univerjs/docs-ui/locale/ru-RU'
import { UniverDrawingPlugin } from '@univerjs/drawing'
import { UniverDrawingUIPlugin } from '@univerjs/drawing-ui'
import DrawingUIruRU from '@univerjs/drawing-ui/locale/ru-RU'
import { UniverRenderEnginePlugin } from '@univerjs/engine-render'
import { UniverUIPlugin } from '@univerjs/ui'
import UIruRU from '@univerjs/ui/locale/ru-RU'

import '@univerjs/design/lib/index.css'
import '@univerjs/ui/lib/index.css'
import '@univerjs/docs-ui/lib/index.css'
import '@univerjs/drawing-ui/lib/index.css'
import '@univerjs-pro/ink-ui/lib/index.css'
import '@univerjs-pro/boards-ui/lib/index.css'

class UniverEditorElement extends HTMLElement {
  private univer?: Univer

  connectedCallback() {
    const container = this

    const univer = new Univer({
      locale: LocaleType.RU_RU,
      locales: {
        [LocaleType.RU_RU]: mergeLocales(
          DesignruRU,
          UIruRU,
          DocsUIruRU,
          DrawingUIruRU,
          InkUIruRU,
          BoardsUIruRU,
        ),
      },
    })

    univer.registerPlugin(UniverLicensePlugin, {
      license: 'your license.txt',
    })
    univer.registerPlugin(UniverRenderEnginePlugin)
    univer.registerPlugin(UniverUIPlugin, { container })
    univer.registerPlugin(UniverDocsPlugin)
    univer.registerPlugin(UniverDocsUIPlugin)
    univer.registerPlugin(UniverDrawingPlugin)
    univer.registerPlugin(UniverDrawingUIPlugin)
    univer.registerPlugin(UniverInkUIPlugin)
    univer.registerPlugin(UniverBoardsPlugin)
    univer.registerPlugin(UniverBoardsUIPlugin)

    univer.createUnit(UniverInstanceType.UNIVER_BOARD, {})
    this.univer = univer
  }

  disconnectedCallback() {
    this.univer?.dispose()
    this.univer = undefined
  }
}

customElements.define('univer-boards-editor', UniverEditorElement)
```

```html
<univer-boards-editor style="display: block; height: 600px"></univer-boards-editor>
```
