---
title: Integrar en Next.js
---

Lea el [instalación y uso básico](/guides/boards/getting-started/installation) primero. Este ejemplo monta las tablas Univer de un cliente de router de aplicaciones Next.js Componente en un contenedor con una altura explícita.

Cargar al editor de un componente de cliente con `ssr: false`[^ssr]. Agregar `use client` no desactiva prerendering por sí mismo. Ver [SSR de esquí en Next.js](https://nextjs.org/docs/app/guides/lazy-loading#skipping-ssr).

Reemplazamiento `YOUR_CLIENT_LICENSE` con el contenido de licencia del cliente, no con una ruta de archivo. [Configuración de la licencia](/server/license).

### Pasos de integración

1. Inicializa Univer en `useEffect`.
2. Libera Univer en `useEffect cleanup`.

### Configuración de plugins [#plugin-configuration]

`UniverUIPlugin`

| <span data-config-options="">Opción</span> | Tipo | Valor predeterminado | Efecto |
| --- | --- | --- | --- |
| `container` | `string \| HTMLElement` |  | Elemento DOM donde se monta el editor o su ID. |

`UniverLicensePlugin`

| <span data-config-options="">Opción</span> | Tipo | Valor predeterminado | Efecto |
| --- | --- | --- | --- |
| `license` | `string` |  | Contenido del archivo de licencia del cliente. |

### Código de ejemplo

Cada montaje crea su propio contenedor hijo. La instancia se libera en una microtarea tras la limpieza del componente React contenedor. Así se evita desmontar de forma síncrona las raíces React anidadas durante un commit.

```tsx title="univer-editor.tsx"
'use client'

import { useEffect, useRef } from 'react'
import { UniverBoardsPlugin } from '@univerjs-pro/boards'
import { UniverBoardsUIPlugin } from '@univerjs-pro/boards-ui'
import BoardsUIesES from '@univerjs-pro/boards-ui/locale/es-ES'
import { UniverInkUIPlugin } from '@univerjs-pro/ink-ui'
import InkUIesES from '@univerjs-pro/ink-ui/locale/es-ES'
import { UniverLicensePlugin } from '@univerjs-pro/license'
import { LocaleType, mergeLocales, Univer, UniverInstanceType } from '@univerjs/core'
import DesignesES from '@univerjs/design/locale/es-ES'
import { UniverDocsPlugin } from '@univerjs/docs'
import { UniverDocsUIPlugin } from '@univerjs/docs-ui'
import DocsUIesES from '@univerjs/docs-ui/locale/es-ES'
import { UniverDrawingPlugin } from '@univerjs/drawing'
import { UniverDrawingUIPlugin } from '@univerjs/drawing-ui'
import DrawingUIesES from '@univerjs/drawing-ui/locale/es-ES'
import { UniverRenderEnginePlugin } from '@univerjs/engine-render'
import { UniverUIPlugin } from '@univerjs/ui'
import UIesES from '@univerjs/ui/locale/es-ES'

import '@univerjs/design/lib/index.css'
import '@univerjs/ui/lib/index.css'
import '@univerjs/docs-ui/lib/index.css'
import '@univerjs/drawing-ui/lib/index.css'
import '@univerjs-pro/ink-ui/lib/index.css'
import '@univerjs-pro/boards-ui/lib/index.css'

export default function UniverEditor() {
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const host = containerRef.current
    if (!host) return
    const container = document.createElement('div')
    container.style.height = '100%'
    host.append(container)

    const univer = new Univer({
      locale: LocaleType.ES_ES,
      locales: {
        [LocaleType.ES_ES]: mergeLocales(
          DesignesES,
          UIesES,
          DocsUIesES,
          DrawingUIesES,
          InkUIesES,
          BoardsUIesES,
        ),
      },
    })

    univer.registerPlugin(UniverLicensePlugin, {
      license: 'YOUR_CLIENT_LICENSE',
    })
    univer.registerPlugin(UniverRenderEnginePlugin)
    univer.registerPlugin(UniverUIPlugin, { container })
    univer.registerPlugin(UniverDocsPlugin)
    univer.registerPlugin(UniverDocsUIPlugin)
    univer.registerPlugin(UniverDrawingPlugin)
    univer.registerPlugin(UniverDrawingUIPlugin)
    univer.registerPlugin(UniverInkUIPlugin)
    univer.registerPlugin(UniverBoardsPlugin)
    univer.registerPlugin(UniverBoardsUIPlugin)

    univer.createUnit(UniverInstanceType.UNIVER_BOARD, {})

    return () => {
      queueMicrotask(() => {
        univer.dispose()
        container.remove()
      })
    }
  }, [])

  return <div ref={containerRef} style={{ height: 600 }} />
}
```

```tsx title="page.tsx"
'use client'

import dynamic from 'next/dynamic'

const UniverEditor = dynamic(() => import('./univer-editor'), { ssr: false })

export default function Page() {
  return <UniverEditor />
}
```

[^ssr]: El renderizado del lado del servidor (SSR) ejecuta el código de renderizado antes de cargar la página en el navegador, sin `window` ni DOM. Un Client Component también puede prerenderizarse, por lo que debe desactivar el SSR del componente del editor.
