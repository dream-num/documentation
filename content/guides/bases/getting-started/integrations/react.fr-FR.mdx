title: Intégrer dans React
---

Cette page commence à partir de [l'installation et l'utilisation de base](/guides/bases/getting-started/installation). Le composant ci-dessous monte Univer Bases dans un conteneur d'une hauteur explicite.

### Étapes d’intégration

1. Initialisez Univer dans `useEffect`.
2. Libérez Univer dans `useEffect cleanup`.

### Configuration des plugins [#plugin-configuration]

`UniverUIPlugin`

| <span data-config-options="">Option</span> | Type | Valeur par défaut | Effet |
| --- | --- | --- | --- |
| `container` | `string \| HTMLElement` |  | Élément DOM de montage de l’éditeur ou son identifiant. |

`UniverLicensePlugin`

| <span data-config-options="">Option</span> | Type | Valeur par défaut | Effet |
| --- | --- | --- | --- |
| `license` | `string` |  | Contenu du fichier de licence côté client. |

### Exemple de code

Chaque montage crée son propre conteneur enfant. L’instance est libérée dans une microtâche après le nettoyage du composant React englobant. Les racines React imbriquées ne sont donc pas démontées de façon synchrone pendant un commit.

```tsx
import { useEffect, useRef } from 'react'
import { UniverBasesPlugin } from '@univerjs-pro/bases'
import { UniverBasesUIPlugin } from '@univerjs-pro/bases-ui'
import BasesUIfrFR from '@univerjs-pro/bases-ui/locale/fr-FR'
import BasesfrFR from '@univerjs-pro/bases/locale/fr-FR'
import { UniverProFormulaEnginePlugin } from '@univerjs-pro/engine-formula'
import { UniverLicensePlugin } from '@univerjs-pro/license'
import { LocaleType, mergeLocales, Univer, UniverInstanceType } from '@univerjs/core'
import DesignfrFR from '@univerjs/design/locale/fr-FR'
import { UniverRenderEnginePlugin } from '@univerjs/engine-render'
import { UniverUIPlugin } from '@univerjs/ui'
import UIfrFR from '@univerjs/ui/locale/fr-FR'

import '@univerjs/design/lib/index.css'
import '@univerjs/ui/lib/index.css'
import '@univerjs-pro/bases-ui/lib/index.css'

export function UniverEditor() {
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const host = containerRef.current
    if (!host) return
    const container = document.createElement('div')
    container.style.height = '100%'
    host.append(container)

    const univer = new Univer({
      locale: LocaleType.FR_FR,
      locales: {
        [LocaleType.FR_FR]: mergeLocales(
          DesignfrFR,
          UIfrFR,
          BasesfrFR,
          BasesUIfrFR,
        ),
      },
    })

    univer.registerPlugin(UniverLicensePlugin, {
      license: 'your license.txt',
    })
    univer.registerPlugin(UniverRenderEnginePlugin)
    univer.registerPlugin(UniverUIPlugin, { container })

    univer.registerPlugin(UniverProFormulaEnginePlugin)
    univer.registerPlugin(UniverBasesPlugin)
    univer.registerPlugin(UniverBasesUIPlugin)

    univer.createUnit(UniverInstanceType.UNIVER_BASE, {})

    return () => {
      queueMicrotask(() => {
        univer.dispose()
        container.remove()
      })
    }
  }, [])

  return <div ref={containerRef} style={{ height: 600 }} />
}
```
