---
title: Integrar en Vue
---

Consulta primero la [instalación y el uso básico](/guides/bases/getting-started/installation). Este ejemplo usa Vue para montar Univer Bases en un contenedor con una altura explícita.

### Pasos de integración

1. Inicializa Univer en `onMounted`.
2. Libera Univer en `onBeforeUnmount`.

### Código de ejemplo

```vue
<script setup lang="ts">
import { onBeforeUnmount, onMounted, ref } from 'vue'
import { UniverBasesPlugin } from '@univerjs-pro/bases'
import { UniverBasesUIPlugin } from '@univerjs-pro/bases-ui'
import BasesUIesES from '@univerjs-pro/bases-ui/locale/es-ES'
import BasesesES from '@univerjs-pro/bases/locale/es-ES'
import { UniverProFormulaEnginePlugin } from '@univerjs-pro/engine-formula'
import { UniverLicensePlugin } from '@univerjs-pro/license'
import { LocaleType, mergeLocales, Univer, UniverInstanceType } from '@univerjs/core'
import DesignesES from '@univerjs/design/locale/es-ES'
import { UniverRenderEnginePlugin } from '@univerjs/engine-render'
import { UniverUIPlugin } from '@univerjs/ui'
import UIesES from '@univerjs/ui/locale/es-ES'

import '@univerjs/design/lib/index.css'
import '@univerjs/ui/lib/index.css'
import '@univerjs-pro/bases-ui/lib/index.css'

const containerRef = ref<HTMLDivElement | null>(null)
let univerInstance: Univer | undefined

onMounted(() => {
  const container = containerRef.value
  if (!container) return

  const univer = new Univer({
    locale: LocaleType.ES_ES,
    locales: {
      [LocaleType.ES_ES]: mergeLocales(
        DesignesES,
        UIesES,
        BasesesES,
        BasesUIesES,
      ),
    },
  })

  univer.registerPlugin(UniverLicensePlugin, {
    license: 'your license.txt',
  })
  univer.registerPlugin(UniverRenderEnginePlugin)
  univer.registerPlugin(UniverUIPlugin, { container })

  univer.registerPlugin(UniverProFormulaEnginePlugin)
  univer.registerPlugin(UniverBasesPlugin)
  univer.registerPlugin(UniverBasesUIPlugin)

  univer.createUnit(UniverInstanceType.UNIVER_BASE, {})
  univerInstance = univer
})

onBeforeUnmount(() => univerInstance?.dispose())
</script>

<template>
  <div ref="containerRef" style="height: 600px" />
</template>
```
