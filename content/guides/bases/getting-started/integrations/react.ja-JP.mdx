---
title: React に統合する
---

まず[インストールと基本的な使い方](/guides/bases/getting-started/installation)を参照してください。以下は React を使い、高さを明示したコンテナーに Univer Bases をマウントする例です。

### 統合手順

1. `useEffect` で Univer を初期化します。
2. `useEffect cleanup` で Univer を破棄します。

### サンプルコード

マウントごとに専用の子コンテナを使用します。外側の React のクリーンアップ後にマイクロタスクで破棄し、コミット中のネストした React root の同期アンマウントを避けます。

```tsx
import { useEffect, useRef } from 'react'
import { UniverBasesPlugin } from '@univerjs-pro/bases'
import { UniverBasesUIPlugin } from '@univerjs-pro/bases-ui'
import BasesUIjaJP from '@univerjs-pro/bases-ui/locale/ja-JP'
import BasesjaJP from '@univerjs-pro/bases/locale/ja-JP'
import { UniverProFormulaEnginePlugin } from '@univerjs-pro/engine-formula'
import { UniverLicensePlugin } from '@univerjs-pro/license'
import { LocaleType, mergeLocales, Univer, UniverInstanceType } from '@univerjs/core'
import DesignjaJP from '@univerjs/design/locale/ja-JP'
import { UniverRenderEnginePlugin } from '@univerjs/engine-render'
import { UniverUIPlugin } from '@univerjs/ui'
import UIjaJP from '@univerjs/ui/locale/ja-JP'

import '@univerjs/design/lib/index.css'
import '@univerjs/ui/lib/index.css'
import '@univerjs-pro/bases-ui/lib/index.css'

export function UniverEditor() {
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const host = containerRef.current
    if (!host) return
    const container = document.createElement('div')
    container.style.height = '100%'
    host.append(container)

    const univer = new Univer({
      locale: LocaleType.JA_JP,
      locales: {
        [LocaleType.JA_JP]: mergeLocales(
          DesignjaJP,
          UIjaJP,
          BasesjaJP,
          BasesUIjaJP,
        ),
      },
    })

    univer.registerPlugin(UniverLicensePlugin, {
      license: 'your license.txt',
    })
    univer.registerPlugin(UniverRenderEnginePlugin)
    univer.registerPlugin(UniverUIPlugin, { container })

    univer.registerPlugin(UniverProFormulaEnginePlugin)
    univer.registerPlugin(UniverBasesPlugin)
    univer.registerPlugin(UniverBasesUIPlugin)

    univer.createUnit(UniverInstanceType.UNIVER_BASE, {})

    return () => {
      queueMicrotask(() => {
        univer.dispose()
        container.remove()
      })
    }
  }, [])

  return <div ref={containerRef} style={{ height: 600 }} />
}
```
