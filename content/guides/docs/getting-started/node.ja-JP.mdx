Univer は同型（isomorphic）特性を備えており、ブラウザだけでなく Node.js 環境でも利用できます。Headless Univer を基にデータ処理サービスを構築し、サーバー側でドキュメントを生成・編集し、Facade API を通じて Univer Docs と連携できます。

Node.js ではプラグインモードを使用し、ブラウザー UI に依存しない必要なプラグインを登録します。

## インストール

```package-install
npm install @univerjs/core @univerjs/engine-render @univerjs/engine-formula @univerjs/docs
```

## 使い方

```typescript
import { LocaleType, Univer } from '@univerjs/core'
import { FUniver } from '@univerjs/core/facade'
import { UniverDocsPlugin } from '@univerjs/docs'
import { UniverFormulaEnginePlugin } from '@univerjs/engine-formula'
import { UniverRenderEnginePlugin } from '@univerjs/engine-render'
import '@univerjs/docs/facade'

const univer = new Univer({
  locale: LocaleType.JA_JP,
  locales: {
    [LocaleType.JA_JP]: {},
  },
})

univer.registerPlugin(UniverRenderEnginePlugin)
univer.registerPlugin(UniverFormulaEnginePlugin)
univer.registerPlugin(UniverDocsPlugin)

const univerAPI = FUniver.newAPI(univer)
univerAPI.createDocument({})
```

## Node.js 環境での共同編集の例 [#node-collaboration-example]

Node.js を共同編集クライアントとして使用し、サーバー上の既存のドキュメントを読み込み、Facade API で内容を変更して、同じドキュメントを開いている他のクライアントに同期できます。実行前に、[共同編集サービス](/server/collaboration/quick-start)、[クライアントライセンス](/server/license)、アクセス可能なドキュメント ID を用意してください。

### サンプルファイル

次の構成でファイルを配置します。

```text
src/docs/node/
├── main.ts
└── univer.ts
```

- `univer.ts`：Univer を作成し、ドキュメント機能のプラグイン、Facade 拡張、Node.js 共同編集クライアントを登録します。
- `main.ts`：共同編集ドキュメントを読み込み、段落、テキスト、ヘッダーとフッター、コールアウト、コードブロック、引用ブロック、リスト、図形、表、段組みの API を順に実行します。

`univer.ts` の `your-license`、サーバー URL、Cookie と、`main.ts` のドキュメント ID を置き換えてください。

import type { FConnectorShape } from '@univerjs-pro/engine-shape/facade'
import { FUniver } from '@univerjs/core/facade'
import process from 'node:process'
import { createUniverOnNode } from './univer'

async function run(): Promise<void> {
  const univer = createUniverOnNode()
  const univerAPI = FUniver.newAPI(univer)

  const document = await univerAPI.getCollaboration().loadDocAsync('your-unit-id')
  if (!document) {
    throw new Error('共同編集ドキュメントを読み込めませんでした。ドキュメント ID とアクセス権限を確認してください。')
  }

  testDocsFacadeAPI(univerAPI)
  testDocsHeaderFooterFacadeAPI(univerAPI)
  testDocsCalloutFacadeAPI(univerAPI)
  testDocsCodeFacadeAPI(univerAPI)
  testDocsQuoteFacadeAPI(univerAPI)
  testDocsListFacadeAPI(univerAPI)
  testDocsShapeFacadeAPI(univerAPI)
  testDocsTableFacadeAPI(univerAPI)
  testDocsColumnFacadeAPI(univerAPI)

  await univerAPI.getCollaboration().flush(document.getId())
  console.warn('ドキュメントの変更をサーバーに同期しました。')
}

function testDocsShapeFacadeAPI(univerAPI: FUniver) {
  const fDocument = univerAPI.getActiveDocument()
  if (!fDocument) return
  const paragraph = fDocument.getParagraphs()[0]
  if (!paragraph) return
  const anchor = {
    paragraphId: paragraph.getId(),
    segmentId: paragraph.getSegmentId(),
  }

  const shape = fDocument.insertShape({
    shapeType: univerAPI.Enum.ShapeTypeEnum.RoundRect,
    placement: {
      wrappingStyle: univerAPI.Enum.TextWrappingStyle.WRAP_SQUARE,
      anchor,
      position: { horizontalOffset: 80, verticalOffset: 80 },
    },
    transform: { width: 320, height: 160 },
    shapeData: {
      fill: {
        fillType: univerAPI.Enum.ShapeFillEnum.SolidFill,
        color: '#DBEAFE',
      },
      stroke: {
        lineStrokeType: univerAPI.Enum.ShapeLineTypeEnum.SolidLine,
        color: '#2563EB',
        width: 2,
      },
    },
  })
  if (!shape) return

  shape
    .setRotation(4)
    .setStrokeOpacity(0.9)
    .getText()
    .setText('ドキュメント図形の Facade サンプル')
    .setHorizontalAlign(univerAPI.Enum.HorizontalAlign.CENTER)
    .setVerticalAlign(univerAPI.Enum.VerticalAlign.MIDDLE)

  const target = fDocument.insertShape({
    shapeType: univerAPI.Enum.ShapeTypeEnum.Ellipse,
    placement: {
      wrappingStyle: univerAPI.Enum.TextWrappingStyle.WRAP_SQUARE,
      anchor,
      position: { horizontalOffset: 500, verticalOffset: 100 },
    },
    transform: { width: 160, height: 120 },
    shapeData: {
      fill: { fillType: univerAPI.Enum.ShapeFillEnum.SolidFill, color: '#DCFCE7' },
      stroke: { lineStrokeType: univerAPI.Enum.ShapeLineTypeEnum.SolidLine, color: '#16A34A', width: 2 },
    },
  })
  if (!target) return

  const connector = fDocument.insertShape({
    shapeType: univerAPI.Enum.ShapeTypeEnum.BentConnector3,
    placement: {
      wrappingStyle: univerAPI.Enum.TextWrappingStyle.WRAP_SQUARE,
      anchor,
      position: { horizontalOffset: 360, verticalOffset: 130 },
    },
    transform: { width: 180, height: 80 },
  })
  if (!connector || !connector.isConnectorShape()) return

  const sourceSites = shape.getConnectionSites()
  const targetSites = target.getConnectionSites()
  ;(connector as FConnectorShape)
    .setStrokeColor('#475569')
    .setStrokeWidth(2)
    .setEndArrow(univerAPI.Enum.ShapeArrowTypeEnum.Arrow, univerAPI.Enum.ShapeArrowSizeEnum.Medium)
    .bindStart(shape.getId(), sourceSites[0]?.index ?? 0)
    .bindEnd(target.getId(), targetSites[2]?.index ?? targetSites[0]?.index ?? 0)

  console.warn('ドキュメント図形 ID：', shape.getId())
  console.warn('ドキュメント図形数：', fDocument.getShapes().length)
}

function testDocsFacadeAPI(univerAPI: FUniver): void {
  const fDocument = univerAPI.getActiveDocument()
  if (!fDocument) return

  const intro = fDocument.appendParagraph(
    `Node.js 共同編集クライアントからのメッセージです。現在の時刻：${new Date().toString()}。`,
  )
  intro.appendText(' [インライン挿入]')

  const beforeIntro = fDocument.insertParagraph(
    intro.getInfo().paragraphIndex,
    '段落オブジェクトを使って紹介段落の前に挿入しました。',
  )
  const foundIntro = fDocument.findParagraphByText('Node.js 共同編集クライアント')
  const introMatches = fDocument.findParagraphs({ text: 'Node.js 共同編集クライアント' })
  const introRange = intro.getRange()
  const bodyBeforeDelete = fDocument.getBody()
  const deleteStartOffset = bodyBeforeDelete.dataStream.indexOf(' [インライン挿入]')
  if (deleteStartOffset >= 0) {
    fDocument.deleteRange({
      startOffset: deleteStartOffset,
      endOffset: deleteStartOffset + ' [インライン挿入]'.length,
    })
  }

  intro.setStyle({
    textStyle: {
      fs: 12,
    },
  })

  console.warn(`現在のドキュメント ID：${fDocument.getId()}`)
  console.warn(`現在のドキュメント名：${fDocument.getName()}`)
  console.warn(`現在のドキュメントがモダンドキュメントか：${fDocument.isModern()}`)
  console.warn(`紹介段落 ID：${intro.getId()}`)
  console.warn(`移動後の紹介段落の子ノードインデックス：${intro.getInfo().paragraphIndex}`)
  console.warn('紹介段落の範囲：', introRange)
  console.warn('見つかった紹介段落 ID：', foundIntro?.getId())
  console.warn('一致する紹介段落数：', introMatches.length)
  console.warn('本文の段落数：', fDocument.getParagraphs().length)
  console.warn(`挿入した段落 ID：${beforeIntro.getId()}`)
  console.warn(`ドキュメントスナップショットの本文の長さ：${fDocument.save().body?.dataStream.length ?? 0}`)
}

function testDocsHeaderFooterFacadeAPI(univerAPI: FUniver): void {
  const fDocument = univerAPI.getActiveDocument()
  if (!fDocument) return

  if (fDocument.isModern()) {
    console.warn(
      'ヘッダーとフッターのサンプルをスキップします。モダンドキュメントはページのヘッダーとフッターに対応していません。',
    )
    return
  }

  const headerSegmentId = fDocument.ensurePageHeader()
  const footerSegmentId = fDocument.ensurePageFooter()

  const headerParagraph = fDocument.appendParagraph('Docs Node.js ヘッダーの Facade サンプル。', headerSegmentId)
  headerParagraph.appendText(` ドキュメント：${fDocument.getName() || fDocument.getId()}`)
  const footerParagraph = fDocument.appendParagraph('Docs Node.js フッターの Facade サンプル。', footerSegmentId)

  fDocument.insertText(0, 'ページヘッダー：', headerSegmentId)
  footerParagraph.setStyle({
    textStyle: {
      fs: 10,
    },
  })

  console.warn('ドキュメントのヘッダーセグメント ID：', headerSegmentId)
  console.warn(
    'ドキュメントのヘッダー段落：',
    fDocument.getParagraphs(headerSegmentId).map((paragraph) => paragraph.getText()),
  )
  console.warn('ドキュメントのヘッダー本文の長さ：', fDocument.getBody(headerSegmentId).dataStream.length)
  console.warn('ドキュメントのフッターセグメント ID：', footerSegmentId)
  console.warn(
    'ドキュメントのフッター段落 ID：',
    fDocument.findParagraphByText('フッターの Facade サンプル', footerSegmentId)?.getId(),
  )
}

function testDocsColumnFacadeAPI(univerAPI: FUniver): void {

        headerRowCount: 1,
        offset: rightOffset,
        tableId: createSampleId('column-table'),

### 実行時の注意点

`loadDocAsync()` はサーバー上の既存ドキュメントを読み込むため、`createDocument()` を呼び出す必要はありません。共同編集サービスは `customHeaders` の Cookie などの認証情報からユーザーを識別し、権限を確認します。接続の `memberID` はサーバーから返されます。

この例ではブラウザのオフラインキャッシュと複数タブ間のインスタンスロックを無効にしているため、アカウントごとのキャッシュ分離を目的とした `setCurrentUser()` の呼び出しは不要です。後からコメントなどローカルの現在のユーザーに依存する機能を使用する場合は、`UserManagerService` で業務ユーザーを設定できます。これはサーバー認証の代わりにはなりません。

ヘッダーとフッターのサンプルはページ分割されたドキュメント、段組みのサンプルはモダンドキュメントでのみ使用できます。サンプルは `isModern()` でドキュメントのモードを判定し、対応していない操作をスキップします。

サンプルは既存ドキュメントに内容を挿入するため、テスト用ドキュメントで実行することをお勧めします。 サンプルは実行後も共同編集接続を維持します。ブラウザで同じドキュメントを開くと変更を確認できます。`flush()` はキュー内の変更に対するサーバーの確認を待ちます。単発のスクリプトにする場合は、すべての非同期処理と同期が完了してから `univer.dispose()` でインスタンスを解放してください。
