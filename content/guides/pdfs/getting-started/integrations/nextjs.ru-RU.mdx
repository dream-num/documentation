---
title: Интеграция с Next.js
---

Сначала прочитайте [установку и базовую настройку](/guides/pdfs/getting-started/installation). В этом примере Next.js App Router монтирует Univer Pdfs в контейнер с явно заданной высотой.

Динамически загрузите редактор из Client Component с `ssr: false`. Директива `use client` сама по себе не отключает предварительный рендеринг. [Next.js](https://nextjs.org/docs/app/guides/lazy-loading#skipping-ssr).

Замените `YOUR_CLIENT_LICENSE` содержимым клиентской лицензии, а не путём к файлу. [Настройка лицензии](/server/license).

### Шаги интеграции

1. Инициализируйте Univer в `useEffect`.
2. Освободите Univer в `useEffect cleanup`.

### Пример кода

Каждое монтирование использует отдельный дочерний контейнер. Освобождение выполняется в микрозадаче после очистки внешнего React, чтобы не размонтировать вложенный React root синхронно во время commit.

```tsx title="app/editor/univer-editor.tsx"
'use client'

import { useEffect, useRef } from 'react'
import { UniverLicensePlugin } from '@univerjs-pro/license'
import { UniverPdfsPlugin } from '@univerjs-pro/pdfs'
import { UniverPdfsUIPlugin } from '@univerjs-pro/pdfs-ui'
import PdfsUIruRU from '@univerjs-pro/pdfs-ui/locale/ru-RU'
import { LocaleType, mergeLocales, Univer } from '@univerjs/core'
import { FUniver } from '@univerjs/core/facade'
import DesignruRU from '@univerjs/design/locale/ru-RU'
import { UniverUIPlugin } from '@univerjs/ui'
import UIruRU from '@univerjs/ui/locale/ru-RU'

import '@univerjs/design/lib/index.css'
import '@univerjs/ui/lib/index.css'
import '@univerjs-pro/pdfs-ui/lib/index.css'

import '@univerjs-pro/pdfs/facade'

export default function UniverEditor() {
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const host = containerRef.current
    if (!host) return
    const container = document.createElement('div')
    container.style.height = '100%'
    host.append(container)

    const univer = new Univer({
      locale: LocaleType.RU_RU,
      locales: {
        [LocaleType.RU_RU]: mergeLocales(
          DesignruRU,
          UIruRU,
          PdfsUIruRU,
        ),
      },
    })

    univer.registerPlugin(UniverLicensePlugin, {
      license: 'YOUR_CLIENT_LICENSE',
    })
    univer.registerPlugin(UniverUIPlugin, { container })
    univer.registerPlugin(UniverPdfsPlugin)
    univer.registerPlugin(UniverPdfsUIPlugin)

    const univerAPI = FUniver.newAPI(univer)
    univerAPI.createPdf({ name: 'Untitled PDF' })

    return () => {
      queueMicrotask(() => {
        univer.dispose()
        container.remove()
      })
    }
  }, [])

  return <div ref={containerRef} style={{ height: 600 }} />
}
```

```tsx title="app/editor/page.tsx"
'use client'

import dynamic from 'next/dynamic'

const UniverEditor = dynamic(() => import('./univer-editor'), { ssr: false })

export default function Page() {
  return <UniverEditor />
}
```
