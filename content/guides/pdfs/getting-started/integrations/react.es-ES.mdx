---
title: Integrar en React
---

Lea [instalación y uso básico](/guides/pdfs/getting-started/installation) primero. El componente React debajo monta Univer Pdfs en un contenedor cuya altura se establece explícitamente.

### Pasos de integración

1. Cree la instancia Univer en `useEffect`.
2. Libera la instancia en la función de limpieza de `useEffect`.

### Configuración de plugins [#plugin-configuration]

`UniverUIPlugin`

| <span data-config-options="">Opción</span> | Tipo | Valor predeterminado | Efecto |
| --- | --- | --- | --- |
| `container` | `string \| HTMLElement` |  | Elemento DOM donde se monta el editor o su ID. |

`UniverLicensePlugin`

| <span data-config-options="">Opción</span> | Tipo | Valor predeterminado | Efecto |
| --- | --- | --- | --- |
| `license` | `string` |  | Contenido del archivo de licencia del cliente. |

### Código de ejemplo

Cada montura obtiene su propio contenedor infantil. La instancia se elimina en un microtarea, después de que la limpieza React circundante haya finalizado, así que las raíces React anidadas nunca se desenrollan sincronizadamente durante un compromiso React.

```tsx
import { useEffect, useRef } from 'react'
import { UniverLicensePlugin } from '@univerjs-pro/license'
import { UniverPdfsPlugin } from '@univerjs-pro/pdfs'
import { UniverPdfsUIPlugin } from '@univerjs-pro/pdfs-ui'
import PdfsUIesES from '@univerjs-pro/pdfs-ui/locale/es-ES'
import { LocaleType, mergeLocales, Univer } from '@univerjs/core'
import { FUniver } from '@univerjs/core/facade'
import DesignesES from '@univerjs/design/locale/es-ES'
import { UniverUIPlugin } from '@univerjs/ui'
import UIesES from '@univerjs/ui/locale/es-ES'

import '@univerjs/design/lib/index.css'
import '@univerjs/ui/lib/index.css'
import '@univerjs-pro/pdfs-ui/lib/index.css'

import '@univerjs-pro/pdfs/facade'

export function UniverEditor() {
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const host = containerRef.current
    if (!host) return
    const container = document.createElement('div')
    container.style.height = '100%'
    host.append(container)

    const univer = new Univer({
      locale: LocaleType.ES_ES,
      locales: {
        [LocaleType.ES_ES]: mergeLocales(
          DesignesES,
          UIesES,
          PdfsUIesES,
        ),
      },
    })

    univer.registerPlugin(UniverLicensePlugin, {
      license: 'your license.txt',
    })
    univer.registerPlugin(UniverUIPlugin, { container })
    univer.registerPlugin(UniverPdfsPlugin)
    univer.registerPlugin(UniverPdfsUIPlugin)

    const univerAPI = FUniver.newAPI(univer)
    univerAPI.createPdf({ name: 'Untitled PDF' })

    return () => {
      queueMicrotask(() => {
        univer.dispose()
        container.remove()
      })
    }
  }, [])

  return <div ref={containerRef} style={{ height: 600 }} />
}
```
