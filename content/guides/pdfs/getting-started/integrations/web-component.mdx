---
title: Integrate in Web Component
---

Read [installation and basic usage](/guides/pdfs/getting-started/installation) first. The custom element below mounts Univer Pdfs in a Web Component, inside a container whose height is set explicitly.

### Integration Steps

1. Create the Univer instance in `connectedCallback`.
2. Dispose of the instance in `disconnectedCallback`.

### Plugin configuration [#plugin-configuration]

`UniverUIPlugin`

| <span data-config-options="">Option</span> | Type | Default | What it does |
| --- | --- | --- | --- |
| `container` | `string \| HTMLElement` |  | Element to mount into, or its DOM ID. |

`UniverLicensePlugin`

| <span data-config-options="">Option</span> | Type | Default | What it does |
| --- | --- | --- | --- |
| `license` | `string` |  | Contents of the client license file. |

### Example

```typescript
import { UniverLicensePlugin } from '@univerjs-pro/license'
import { UniverPdfsPlugin } from '@univerjs-pro/pdfs'
import { UniverPdfsUIPlugin } from '@univerjs-pro/pdfs-ui'
import PdfsUIenUS from '@univerjs-pro/pdfs-ui/locale/en-US'
import { LocaleType, mergeLocales, Univer } from '@univerjs/core'
import { FUniver } from '@univerjs/core/facade'
import DesignenUS from '@univerjs/design/locale/en-US'
import { UniverUIPlugin } from '@univerjs/ui'
import UIenUS from '@univerjs/ui/locale/en-US'

import '@univerjs/design/lib/index.css'
import '@univerjs/ui/lib/index.css'
import '@univerjs-pro/pdfs-ui/lib/index.css'

import '@univerjs-pro/pdfs/facade'

class UniverEditorElement extends HTMLElement {
  private univer?: Univer

  connectedCallback() {
    const container = this

    const univer = new Univer({
      locale: LocaleType.EN_US,
      locales: {
        [LocaleType.EN_US]: mergeLocales(
          DesignenUS,
          UIenUS,
          PdfsUIenUS,
        ),
      },
    })

    univer.registerPlugin(UniverLicensePlugin, {
      license: 'your license.txt',
    })
    univer.registerPlugin(UniverUIPlugin, { container })
    univer.registerPlugin(UniverPdfsPlugin)
    univer.registerPlugin(UniverPdfsUIPlugin)

    const univerAPI = FUniver.newAPI(univer)
    univerAPI.createPdf({ name: 'Untitled PDF' })
    this.univer = univer
  }

  disconnectedCallback() {
    this.univer?.dispose()
    this.univer = undefined
  }
}

customElements.define('univer-pdfs-editor', UniverEditorElement)
```

```html
<univer-pdfs-editor style="display: block; height: 600px"></univer-pdfs-editor>
```
