---
title: Integrate in Next.js
---

Start with [installation and basic usage](/guides/pdfs/getting-started/installation). The example below uses the Next.js App Router to mount Univer Pdfs in a container with an explicit height.

Load the editor from a Client Component with `ssr: false`[^ssr]. A `use client` directive alone does not disable prerendering. See [Next.js](https://nextjs.org/docs/app/guides/lazy-loading#skipping-ssr).

Place the client license content where `YOUR_CLIENT_LICENSE` appears; a file path does not work here. See [License configuration](/server/license).

### Integration steps

1. Create the Univer instance in `useEffect`.
2. Dispose of the instance in `useEffect cleanup`.

### Plugin configuration [#plugin-configuration]

`UniverUIPlugin`

| <span data-config-options="">Option</span> | Type | Default | What it does |
| --- | --- | --- | --- |
| `container` | `string \| HTMLElement` |  | Mount element or its DOM ID. |

`UniverLicensePlugin`

| <span data-config-options="">Option</span> | Type | Default | What it does |
| --- | --- | --- | --- |
| `license` | `string` |  | Client license file contents. |

### Example

Each mount gets its own child container. Disposal runs in a microtask after the surrounding React cleanup, so nested React roots are not unmounted synchronously during a React commit.

```tsx title="univer-editor.tsx"
'use client'

import { useEffect, useRef } from 'react'
import { UniverLicensePlugin } from '@univerjs-pro/license'
import { UniverPdfsPlugin } from '@univerjs-pro/pdfs'
import { UniverPdfsUIPlugin } from '@univerjs-pro/pdfs-ui'
import PdfsUIenUS from '@univerjs-pro/pdfs-ui/locale/en-US'
import { LocaleType, mergeLocales, Univer } from '@univerjs/core'
import { FUniver } from '@univerjs/core/facade'
import DesignenUS from '@univerjs/design/locale/en-US'
import { UniverUIPlugin } from '@univerjs/ui'
import UIenUS from '@univerjs/ui/locale/en-US'

import '@univerjs/design/lib/index.css'
import '@univerjs/ui/lib/index.css'
import '@univerjs-pro/pdfs-ui/lib/index.css'

import '@univerjs-pro/pdfs/facade'

export default function UniverEditor() {
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const host = containerRef.current
    if (!host) return
    const container = document.createElement('div')
    container.style.height = '100%'
    host.append(container)

    const univer = new Univer({
      locale: LocaleType.EN_US,
      locales: {
        [LocaleType.EN_US]: mergeLocales(
          DesignenUS,
          UIenUS,
          PdfsUIenUS,
        ),
      },
    })

    univer.registerPlugin(UniverLicensePlugin, {
      license: 'YOUR_CLIENT_LICENSE',
    })
    univer.registerPlugin(UniverUIPlugin, { container })
    univer.registerPlugin(UniverPdfsPlugin)
    univer.registerPlugin(UniverPdfsUIPlugin)

    const univerAPI = FUniver.newAPI(univer)
    univerAPI.createPdf({ name: 'Untitled PDF' })

    return () => {
      queueMicrotask(() => {
        univer.dispose()
        container.remove()
      })
    }
  }, [])

  return <div ref={containerRef} style={{ height: 600 }} />
}
```

```tsx title="page.tsx"
'use client'

import dynamic from 'next/dynamic'

const UniverEditor = dynamic(() => import('./univer-editor'), { ssr: false })

export default function Page() {
  return <UniverEditor />
}
```

[^ssr]: Server-side rendering (SSR) runs rendering code before the browser loads the page. Browser-only APIs such as `window` and the DOM are unavailable there; a Client Component may still be prerendered unless SSR is disabled for that editor component.
