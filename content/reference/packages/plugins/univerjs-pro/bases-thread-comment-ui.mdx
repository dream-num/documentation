```typescript
import enUS from '@univerjs-pro/bases-thread-comment-ui/locale/en-US'

import '@univerjs-pro/bases-thread-comment-ui/lib/index.css'

univer.registerPlugin(UniverBasesThreadCommentUIPlugin)
```

- Locale: `@univerjs-pro/bases-thread-comment-ui/locale/*`
- Styles: `@univerjs-pro/bases-thread-comment-ui/lib/index.css`
