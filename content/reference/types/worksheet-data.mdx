The data for one worksheet: cells, row and column sizes, merged ranges, and display settings. Workbooks store sheets under `IWorkbookData.sheets`; each key matches the corresponding sheet ID in `sheetOrder`.

{/* API details are generated by scripts/sync-api-reference.mjs. */}

## Import

```ts
import type { IWorksheetData } from '@univerjs/core'
```

## Definition

```ts
interface IWorksheetData {
  id: string

  name: string
  tabColor: string

  hidden: WorksheetHiddenState | BooleanNumber
  freeze: IFreeze
  rowCount: number
  columnCount: number
  zoomRatio: number
  scrollTop: number
  scrollLeft: number
  defaultColumnWidth: number
  defaultRowHeight: number

  mergeData: IRange[]

  cellData: IObjectMatrixPrimitiveType<ICellData>
  rowData: IObjectArrayPrimitiveType<Partial<IRowData>>
  columnData: IObjectArrayPrimitiveType<Partial<IColumnData>>

  defaultStyle?: Nullable<IStyleData> | string
  rowHeader: {
    width: number
    hidden?: BooleanNumber
  }
  columnHeader: {
    height: number
    hidden?: BooleanNumber
  }
  showGridlines: BooleanNumber

  gridlinesColor?: string

  backgroundImage?: IWorksheetBackgroundImage
  rightToLeft: BooleanNumber

  custom?: CustomData
}
```

### `hidden` [#hidden]

Required · `BooleanNumber | WorksheetHiddenState`

Sheet hiding state: 0 = visible, 1 = hidden, 2 = very hidden.
BooleanNumber remains accepted for existing 0/1 snapshots and integrations.
Use WorksheetHiddenState for new code; do not interpret this as a boolean flag.
